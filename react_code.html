1. index.js

📁 /index.js
หน้าที่: จุดเริ่มต้นของ App ทั้งหมด — ลงทะเบียน Root Component กับ React Native
import { AppRegistry } from 'react-native'; import App from './App'; import { name as appName } from './app.json'; AppRegistry.registerComponent(appName, () => App);
💡 อธิบาย:
  • AppRegistry.registerComponent — บอก React Native ว่า Component ไหนเป็น "ตัวหลัก" ที่จะแสดงผล
  • appName — ชื่อ App ดึงจาก app.json (ค่า = "ReactNativeApp")
  • () => App — ส่ง function ที่ return App component
  • ไฟล์นี้ถูกเรียกโดย Native code (MainActivity.kt บน Android) เมื่อเปิด App

2. App.tsx

📁 /App.tsx
หน้าที่: Component หลัก — เตรียมฐานข้อมูล, ครอบ Provider, แสดง Navigation
const App: React.FC = () => { const [ready, setReady] = useState(false); useEffect(() => { const init = async () => { await DatabaseService.getInstance().initialize(); setReady(true); }; init(); }, []); if (!ready) return <ActivityIndicator />; return ( <PaperProvider> <NavigationContainer> <AppNavigator /> </NavigationContainer> <Toast /> </PaperProvider> ); };
💡 อธิบายทีละบรรทัด:
  • useState(false) — สร้าง state "ready" เริ่มต้น = false (ยังไม่พร้อม)
  • useEffect(..., []) — ทำงานครั้งเดียวตอน App เปิด (เหมือน onMount)
  • DatabaseService.getInstance().initialize() — เปิดฐานข้อมูล + สร้างตาราง
  • setReady(true) — เปลี่ยน state เป็น "พร้อม" → แสดงหน้าจอหลัก
  • PaperProvider — ครอบทั้ง App เพื่อให้ใช้ UI components ของ react-native-paper ได้
  • NavigationContainer — ครอบระบบ navigation ทั้งหมด
  • Toast — component สำหรับแสดงข้อความแจ้งเตือน (ต้องอยู่นอก NavigationContainer)

3. types/index.ts

📁 /src/types/index.ts
หน้าที่: กำหนดรูปร่างข้อมูล (TypeScript Interface) — บอก compiler ว่าข้อมูลแต่ละตัวมี field อะไรบ้าง
export interface Product { id: number; // รหัสสินค้า (auto-increment) name: string; // ชื่อสินค้า unit: string; // หน่วยนับ เช่น "ชิ้น" category: 'ไฟฟ้า' | 'Accessory' | 'อิเล็กทรอนิกส์' | 'พลาสติก'; description: string; // รายละเอียด reorderPoint: number; // จุดสั่งซื้อใหม่ (0 = ปิด) } export interface StockEntry { productId: number; // อ้างอิงสินค้า zone: string; // โซน เช่น "A1" quantity: number; // จำนวนคงเหลือ } export interface TransactionLog { id: number; type: 'inbound' | 'outbound' | 'transfer'; productId: number; fromZone: string | null; toZone: string | null; quantity: number; timestamp: string; } export interface DashboardData { totalStock: number; zoneAStock: number; zoneBStock: number; zoneCStock: number; totalProducts: number; totalTransactions: number; }
💡 ทำไมต้องมี Types?
  • ช่วยให้ editor แนะนำ (autocomplete) ได้ถูกต้อง
  • จับ bug ตั้งแต่ตอนเขียน — ถ้าพิมพ์ชื่อ field ผิดจะ error ทันที
  • เป็นเอกสารของโค้ด — อ่านแล้วรู้เลยว่าข้อมูลหน้าตาเป็นยังไง

5. DatabaseService.ts

📁 /src/db/DatabaseService.ts
หน้าที่: จัดการฐานข้อมูลทั้งหมด — CRUD สินค้า, จัดการ stock, บันทึก log
📐 Design Pattern: Singleton

มี instance เดียวตลอด App — ทุกหน้าจอเรียกใช้ DB ตัวเดียวกัน

// เรียกใช้จากทุกที่: const db = DatabaseService.getInstance();
📦 Methods สำคัญ:
  • initialize() เปิด DB + สร้างตาราง
  • getAllProducts() ดึงสินค้าทั้งหมด
  • addProduct() เพิ่มสินค้าใหม่
  • updateProduct() แก้ไขสินค้า
  • deleteProduct() ลบสินค้า
  • inbound() รับสินค้าเข้า → เพิ่ม stock + บันทึก log
  • transfer() ย้ายสินค้า → ลด stock ต้นทาง + เพิ่ม stock ปลายทาง + log
  • outbound() จ่ายออก → ลด stock + log
  • getDashboardSummary() สรุปข้อมูลหน้า Dashboard
  • getLowStockProducts() สินค้าที่ต่ำกว่า reorder point
🔒 Transaction คืออะไร?

ใช้ db.transaction() เพื่อให้หลาย SQL ทำงานเป็น "ชุดเดียว" — ถ้า query ใดพัง ทั้งชุดจะถูก rollback (ย้อนกลับ) ป้องกันข้อมูลไม่สมบูรณ์

// ตัวอย่าง: inbound — ทำ 2 อย่างพร้อมกัน await db.transaction(async (tx) => { // 1. เพิ่ม stock (ถ้ามีอยู่แล้วก็บวกเพิ่ม) await tx.execute( `INSERT INTO stock ... ON CONFLICT DO UPDATE SET quantity = quantity + ?` ); // 2. บันทึกประวัติ await tx.execute( `INSERT INTO logs ...` ); });
